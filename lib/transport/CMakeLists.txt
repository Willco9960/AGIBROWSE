if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Task 008 transport requires Windows/MSVC; platform parity remains deferred")
endif()
set(AGI_TRANSPORT_DEPS "${CMAKE_SOURCE_DIR}/build/transport-deps" CACHE PATH "Verified transport source/build cache")
set(OPENSSL_ROOT_DIR "${AGI_TRANSPORT_DEPS}/openssl-install")
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(OPENSSL_MSVC_STATIC_RT TRUE)
if(NOT EXISTS "${OPENSSL_ROOT_DIR}/agi-build.json")
  message(FATAL_ERROR "Complete verified static transport dependency receipt missing; run tools/transport/bootstrap.ps1")
endif()
file(READ "${OPENSSL_ROOT_DIR}/agi-build.json" AGI_OPENSSL_RECEIPT)
file(SHA256 "${CMAKE_SOURCE_DIR}/tools/transport/dependencies.lock.json" AGI_LOCK_DIGEST)
string(JSON AGI_RECEIPT_LOCK GET "${AGI_OPENSSL_RECEIPT}" lockSha256)
string(TOLOWER "${AGI_RECEIPT_LOCK}" AGI_RECEIPT_LOCK)
if(NOT AGI_RECEIPT_LOCK STREQUAL AGI_LOCK_DIGEST)
  message(FATAL_ERROR "Transport dependency receipt does not match the exact lock file")
endif()
find_package(OpenSSL 3.5.9 EXACT REQUIRED)
if(NOT EXISTS "${AGI_TRANSPORT_DEPS}/boost_1_92_0/boost/beast.hpp")
  message(FATAL_ERROR "Run tools/transport/bootstrap.ps1 before building")
endif()
add_library(agi-transport STATIC pairing.cpp server.cpp)
target_compile_features(agi-transport PUBLIC cxx_std_20)
target_compile_options(agi-transport PUBLIC /EHsc /bigobj)
target_include_directories(agi-transport PUBLIC ${CMAKE_SOURCE_DIR} "${AGI_TRANSPORT_DEPS}/boost_1_92_0")
target_compile_definitions(agi-transport PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX BOOST_ASIO_NO_DEPRECATED BOOST_ERROR_CODE_HEADER_ONLY _WIN32_WINNT=0x0A00)
target_link_libraries(agi-transport PUBLIC OpenSSL::SSL OpenSSL::Crypto crypt32 comdlg32 ws2_32)
set_target_properties(agi-transport PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(agi-pair-client ../../apps/broker/src/pair_client.cpp)
target_link_libraries(agi-pair-client PRIVATE agi-transport)
set_target_properties(agi-pair-client PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(BUILD_TESTING)
  add_executable(agi-transport-tests ../../tests/security/transport_native.cpp)
  target_link_libraries(agi-transport-tests PRIVATE agi-transport agi-private-ipc)
  set_target_properties(agi-transport-tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME authenticated-loopback-transport COMMAND agi-transport-tests)
  set_tests_properties(authenticated-loopback-transport PROPERTIES TIMEOUT 120)
endif()
