add_library(agi-privacy STATIC publication.cpp)
target_include_directories(agi-privacy PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../..)
target_compile_features(agi-privacy PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(agi-privacy PUBLIC /EHsc)
endif()
set_target_properties(agi-privacy PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(BUILD_TESTING)
  add_executable(agi-privacy-tests ../../tests/security/privacy_native.cpp)
  target_link_libraries(agi-privacy-tests PRIVATE agi-privacy)
  set_target_properties(agi-privacy-tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME privacy-publication-contract COMMAND agi-privacy-tests)
  if(AGI_BROWSE_WITH_TRANSPORT)
    add_executable(agi-privacy-transport-tests ../../tests/security/privacy_transport_native.cpp)
    target_link_libraries(agi-privacy-transport-tests PRIVATE agi-transport)
    set_target_properties(agi-privacy-transport-tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_test(NAME privacy-actual-broker-export COMMAND agi-privacy-transport-tests $<TARGET_FILE:agi-browse-broker>)
    set_tests_properties(privacy-actual-broker-export PROPERTIES TIMEOUT 45)
  endif()
endif()
